Linker script

-T memmap.ld

group names, page names and bank names cannot overlap

###############################################################################
# MEMORY BANKS
###############################################################################

BANK FIXED
    BASE 0x0000
    SIZE 0x4000

    HOLE IOREG     AT 0x0000 SIZE 0x0100
    RESERVE BOOT   AT 0x0100 SIZE 0x0100


BANK SWITCHED
    BASE 0xC000
    SIZE 0x4000


###############################################################################
# BANK SWITCHING PAGES
###############################################################################

PAGE PAGE0 IN SWITCHED
PAGE PAGE1 IN SWITCHED
PAGE PAGE2 IN SWITCHED


###############################################################################
# GROUP DEFINITIONS
###############################################################################

# Code group (uncompressed)
GROUP CODE MAXSIZE 0x3000 : code_* ""

# Data group
GROUP DATA MAXSIZE 0x2000 : data_*

# BSS group
GROUP BSS  MAXSIZE 0x1000 : bss_*

# Interrupt vectors
GROUP INT  MAXSIZE 0x0100 : int_*

# Level data (to be banked)
GROUP LEVEL1 MAXSIZE 0x2000 : level1_*
GROUP LEVEL2 MAXSIZE 0x2000 : level2_*
GROUP LEVEL3 MAXSIZE 0x2000 : level3_*


###############################################################################
# COMPRESSION RULES
###############################################################################

# Compress CODE into CODE_C, expand at runtime to 0x2000
COMPRESS CODE USING zx7
    STORED_AS CODE_C
    EXPANDER zx7_expand
    EXPANDS_TO 0x2000


###############################################################################
# PLACEMENT RULES
###############################################################################

# Place compressed CODE in FIXED bank after reserved areas
PLACE CODE_C IN FIXED ALIGN 0x100 FILL 0xFF

# Place DATA after CODE_C
PLACE DATA AFTER CODE_C ALIGN 0x100 FILL 0x00

# Place INT at absolute address
PLACE INT AT 0x0000

# Place BSS within a subrange of FIXED
PLACE BSS WITHIN 0x3000 SIZE 0x0800

# Place banked level data into pages
PLACE LEVEL1 IN PAGE0
PLACE LEVEL2 IN PAGE1
PLACE LEVEL3 IN PAGE2


###############################################################################
# SYMBOL DEFINITIONS
###############################################################################

# Bank symbols
SYMBOL fixed_start = FIXED.start
SYMBOL fixed_end   = FIXED.end

# Group symbols
SYMBOL code_expanded_start = CODE.start
SYMBOL code_expanded_end   = CODE.end

# Offsets
SYMBOL heap_start = DATA.end + 0x10
SYMBOL stack_top  = FIXED.end - 0x20



BANK <name>
    BASE <addr>
    SIZE <size>

    HOLE    <name> AT <addr> SIZE <size>    ; hardware hole
    RESERVE <name> AT <addr> SIZE <size>    ; reserve memory

GROUP <name> MAXSIZE <size> : code_* ""

COMPRESS CODE USING zx7
    STORED_AS CODE_C
    EXPANDER zx7_expand
    EXPANDS_TO 0x2000

; USING tool compresses a binary file
; EXPANDER expands the compressed data, 
; source address of compressed data in HL and the target address in DE

GROUP <name> MAXSIZE <size> : data_*
GROUP <name> MAXSIZE <size> : bss_*
GROUP <name> MAXSIZE <size> : int_*

PAGE <name> IN <bank>

PLACE <group> IN <bank>     ALIGN <align> FILL <fill>
PLACE <group> AFTER <group> ALIGN <align> FILL <fill>
PLACE <group> IN <page>     ALIGN <align> FILL <fill>
PLACE <group> AT <addr>
PLACE <group> WITHIN <addr> SIZE <size>
PLACE CODE_C  IN BANK1

SYMBOL <name> = <bank>.[start|end]  [+-] <offset>
SYMBOL <name> = <group>.[start|end] [+-] <offset>

bank switching


BANK COMMON 0x0000 SIZE 16K

BANK GAME0  0x4000 SIZE 16K
BANK GAME1  0x4000 SIZE 16K
BANK GAME2  0x4000 SIZE 16K

BANKSWITCH PORT 0x7F

TRAMPOLINES IN COMMON

PLACE CORE     IN COMMON
PLACE GRAPHICS IN GAME0
PLACE AUDIO    IN GAME1
PLACE AI       IN GAME2

AUTO FARCALLS


BANK COMMON 0x0000 SIZE 16K

BANK GAME0 NUMBER 0
BANK GAME1 NUMBER 1
BANK GAME2 NUMBER 2

FARCALLS USING __bank_call

Then the linker generates calls to __bank_call, passing the destination bank and address according to a documented ABI.
That keeps the linker portable across virtually every Z80 banking scheme while still providing all the value of automatic trampoline generation. In my view, that's the cleanest separation of responsibilities.
